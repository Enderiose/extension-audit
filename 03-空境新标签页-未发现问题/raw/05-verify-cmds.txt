4) 包完整性：按 Chromium treehash 复算商店签名清单
────────────────────────────────────────────────────────────
python - <<'PY'
import json,base64,hashlib
d=json.load(open('ext/_metadata/verified_contents.json'))
p=json.loads(base64.urlsafe_b64decode(d[0]['signed_content']['payload']+'==='))
bs=p['content_hashes'][0]['block_size']
b=lambda x: base64.urlsafe_b64encode(x).decode().rstrip('=')
ok=bad=0
for f in p['content_hashes'][0]['files']:
    data=open('ext/'+f['path'],'rb').read()
    hs=[hashlib.sha256(data[i:i+bs]).digest() for i in range(0,len(data),bs)]
    root=b(hs[0]) if len(hs)==1 else b(hashlib.sha256(b''.join(hs)).digest())
    ok+= root==f['root_hash']; bad+= root!=f['root_hash']
print(p['item_id'],p['item_version'],'ok',ok,'bad',bad)

import zipfile,io,re,collections

 '仿冒域': r'\b[a-z0-9\-]+\.(?:coml|cm|con|orn|om)\.(?:cc|cn|com|net)\b',
 '推广码参数': r'\b(?:tn|pid|unionid|union_id|aff|affid|cps|ref|spm|scm|ali_refid|ali_trackid)=',
 '点击/跳转路径': r'(/clk|/click|/track|/redirect|/jump|/goto|/union|/cps|/tg\?)',
 '指纹采集': r'(fingerprint|canvas\.toDataURL|webgl|deviceId|fp_?id)',
 '动态执行': r'(eval\s*\(|new\s+Function\s*\(|atob\s*\()',
